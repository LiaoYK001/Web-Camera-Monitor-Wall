cmake_minimum_required(VERSION 3.28)
project(webobs-native-host VERSION 0.1.0 LANGUAGES CXX)

option(WEBOBS_NATIVE_BUILD_PLUGIN "Build the actual OBS/Qt frontend plugin" ON)
option(WEBOBS_NATIVE_BUILD_TESTS "Build API-contract tests (no renderer/decoder)" OFF)
option(WEBOBS_NATIVE_PIXEL_TEST "NON-REDISTRIBUTABLE synthetic final-window pixel test entry point" OFF)
if(WEBOBS_NATIVE_PIXEL_TEST AND NOT WEBOBS_NATIVE_BUILD_PLUGIN)
  message(FATAL_ERROR "The pixel harness requires the actual OBS frontend plugin")
endif()
set(WEBOBS_NATIVE_OBS_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../obs/obs-studio" CACHE PATH "Pinned OBS source headers for tests")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(WEBOBS_NATIVE_BUILD_PLUGIN)
  find_package(libobs 32.1.2 EXACT CONFIG REQUIRED)
  find_package(obs-frontend-api 32.1.2 EXACT CONFIG REQUIRED)
  find_package(Qt6 CONFIG REQUIRED COMPONENTS Widgets)
  add_library(webobs-native-host MODULE src/plugin.cpp src/wall.cpp src/product_bootstrap.cpp src/config_backup.cpp)
  if(WEBOBS_NATIVE_PIXEL_TEST)
    target_sources(webobs-native-host PRIVATE src/pixel_test.cpp)
    target_compile_definitions(webobs-native-host PRIVATE WEBOBS_NATIVE_PIXEL_TEST=1)
    install(FILES tests/pixel-test-build.json DESTINATION . RENAME webobs-native-pixel-test-build.json)
  endif()
  target_link_libraries(webobs-native-host PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Widgets)
  set_target_properties(webobs-native-host PROPERTIES PREFIX "")
  if(MSVC)
    target_compile_options(webobs-native-host PRIVATE /W4 /utf-8)
  else()
    target_compile_options(webobs-native-host PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  if(WIN32)
    install(TARGETS webobs-native-host LIBRARY DESTINATION obs-plugins/64bit RUNTIME DESTINATION obs-plugins/64bit)
  else()
    install(TARGETS webobs-native-host LIBRARY DESTINATION lib/obs-plugins)
  endif()
  install(DIRECTORY data/ DESTINATION data/obs-plugins/webobs-native-host)
endif()

if(WEBOBS_NATIVE_BUILD_TESTS)
  enable_testing()
  # Pinned OBS uses the SDK SSE2 intrinsics on MSVC x64; other targets use SIMDe.
  set(native_test_simde_include "")
  if(NOT (MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 8 AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64|X86_64)$"))
    find_path(SIMDE_INCLUDE_DIR simde/x86/sse2.h REQUIRED)
    set(native_test_simde_include "${SIMDE_INCLUDE_DIR}")
  endif()
  if(NOT EXISTS "${WEBOBS_NATIVE_OBS_SOURCE}/libobs/obs.h")
    message(FATAL_ERROR "Tests require the checked-out pinned OBS headers")
  endif()
  # These declarations only need configuration macros, not an installed libobs.
  # No old system/local OBS binary is linked and no fake rendering is claimed.
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/test-config")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/test-config/obsconfig.h"
    "#pragma once\n#define OBS_DATA_PATH \"\"\n#define OBS_INSTALL_PREFIX \"\"\n#define OBS_RELEASE_CANDIDATE 0\n#define OBS_BETA 0\n")
  add_executable(native-wall-test tests/wall_test.cpp src/wall.cpp)
  add_executable(native-product-bootstrap-test tests/product_bootstrap_test.cpp src/product_bootstrap.cpp)
  foreach(test_target IN ITEMS native-wall-test native-product-bootstrap-test)
    target_include_directories(${test_target} PRIVATE src "${WEBOBS_NATIVE_OBS_SOURCE}/libobs"
      "${WEBOBS_NATIVE_OBS_SOURCE}/frontend/api" "${CMAKE_CURRENT_BINARY_DIR}/test-config" ${native_test_simde_include})
    if(MSVC)
      target_compile_options(${test_target} PRIVATE /W4 /UNDEBUG /utf-8)
    else()
      target_compile_options(${test_target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    endif()
  endforeach()
  add_test(NAME native-wall-api-contract COMMAND native-wall-test)
  add_test(NAME native-product-bootstrap-api-contract COMMAND native-product-bootstrap-test)
  set_tests_properties(native-wall-api-contract native-product-bootstrap-api-contract
    PROPERTIES TIMEOUT 30 LABELS "native;api-contract;no-media")
endif()
